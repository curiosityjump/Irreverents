import { canAct } from '../body-state.ts';
import { drive, isOut } from '../npc/state.ts';
import { outposts } from '../outpost/state.ts';
import { BREAKABLE, breakPartAt } from '../room-parts.ts';
import { partState } from '../room-state.ts';
import type { System } from '../types.ts';

const EPS = 1e-6;

/**
 * The orchestrator, 2026-11-24, on the owner's "come in from the sides" in the
 * Bandit Camp's east half (cut into pockets by pieces the camp's people cannot
 * climb): a row walks in off screen, and a wave's bandit blocked by a breakable
 * piece between him and the hero he is coming for breaks it, a short wind-up
 * and then the piece's own broken stage. DRAFT (docs/decisions-log.md "The
 * camp's rows break through"): only an outpost with `breakThrough` (the camp;
 * the depot and the test rooms leave it out), only a wave's person (placed
 * people never do), only while he walks at the place he was sent to or saw a
 * hero (`outpostFight`), and only a piece no tougher than `upTo` px/s: a
 * tougher one (the camp's barricade) still stops him and stays cover.
 * INTENT: impact over numbers (D-020): he breaks it through the breakables'
 * own path (breakPartAt, room-parts.ts); the piece has no health.
 *
 * Order 60: after outpost-fight (58) has set his walk for this tick, before
 * outpost-leave (61) and physics (500). The wind-up holds attack, so his swing
 * is drawn while he winds up.
 */
const system: System = {
  id: 'outpost-break-through',
  order: 60,
  run(ctx) {
    const { world, content } = ctx;
    for (const op of outposts(content)) {
      const bt = op.breakThrough;
      if (!bt) continue;
      const room = content.rooms[op.room];
      if (!room) continue;
      const T = room.tile;
      const windup = Math.max(1, Math.round(bt.time * 60));
      for (const e of world.entities) {
        const w = e.c.outpostWave;
        if (!w || w.outpost !== op.id || e.room !== op.room || e.player !== null) continue;
        const target = e.c.outpostFight;
        const box = content.characters[e.kind]?.box;
        const dir = Math.sign(e.c.input?.x ?? 0);
        const seated = world.entities.some((m) => (m.c as { seat?: { driver: string | null } }).seat?.driver === e.id);
        if (isOut(e) || !canAct(e) || seated || w.going || !target || !box || dir === 0 || Math.sign(target.x - (e.x + box.w / 2)) !== dir) {
          w.breaking = 0;
          continue;
        }
        // The tile column just ahead of him, at his body's rows.
        const tx = dir > 0 ? Math.floor((e.x + box.w + 1 - EPS) / T) : Math.floor((e.x - 1) / T);
        const top = Math.floor(e.y / T);
        const bottom = Math.floor((e.y + box.h - EPS) / T);
        let at: number | null = null;
        for (let ty = top; ty <= bottom && at === null; ty++) {
          const part = room.parts.find((p) => p.kind === BREAKABLE && p.tiles.some((t) => t.tx === tx && t.ty === ty));
          if (!part || partState(world, room.id, part.id).broken) continue;
          const speed = typeof part.params.breakSpeed === 'number' ? part.params.breakSpeed : content.tuning.world.wallBreakSpeed!;
          if (speed <= bt.upTo) at = ty;
        }
        if (at === null) {
          w.breaking = 0;
          continue;
        }
        w.breaking = (w.breaking ?? 0) + 1;
        drive(e, { x: 0, y: 0, held: ['attack'] });
        if (w.breaking >= windup) {
          breakPartAt(ctx, room.id, tx, at, e);
          w.breaking = 0;
        }
      }
    }
  },
};
export default system;
